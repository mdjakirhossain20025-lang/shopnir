১. দুইটি সংখ্যার যোগ, বিয়োগ, গুণ ও ভাগ

এটি beginner-দের জন্য খুব গুরুত্বপূর্ণ একটি program। এখানে input(), float(), arithmetic operator এবং variable—সবগুলো একসঙ্গে ব্যবহার করা হয়েছে।

num1 = float(input("প্রথম সংখ্যা লিখুন: ")) num2 = float(input("দ্বিতীয় সংখ্যা লিখুন: ")) addition = num1 + num2 subtraction = num1 - num2 multiplication = num1 * num2 print("যোগ =", addition) print("বিয়োগ =", subtraction) print("গুণ =", multiplication) if num2 != 0: print("ভাগ =", num1 / num2) else: print("শূন্য দিয়ে ভাগ করা যাবে না।")
উদাহরণ:

বিয়োগ = 15
গুণ = 100

২. জোড় ও বিজোড় সংখ্যা নির্ণয়

number = int(input("একটি পূর্ণসংখ্যা দিন: ")) if number % 2 == 0: print(number, "জোড় সংখ্যা।") else: print(number, "বিজোড় সংখ্যা।")
% operator ভাগশেষ বের করে। তাই number % 2 == 0 মানে সংখ্যাটি 2 দিয়ে নিঃশেষে বিভাজ্য।

৩. ১ থেকে N পর্যন্ত যোগফল

n = int(input("শেষ সংখ্যা দিন: ")) total = 0 for number in range(1, n + 1): total = total + number print("মোট যোগফল:", total)

৫. Factorial Program

number = int(input("একটি সংখ্যা দিন: ")) factorial = 1 for i in range(1, number + 1): factorial = factorial * i print(number, "এর factorial =", factorial)

৬. Prime Number পরীক্ষা

number = int(input("একটি সংখ্যা দিন: ")) if number < 2: print("Prime নয়।") else: is_prime = True for i in range(2, number): if number % i == 0: is_prime = False break if is_prime: print(number, "Prime number।") else: print(number, "Prime নয়।")
break ব্যবহার করা হয়েছে কারণ divisor পাওয়া গেলে আর loop চালানোর প্রয়োজন নেই।
Practice: 1 থেকে 100 পর্যন্ত সব prime number বের করুন।

def get_grade(marks): if marks >= 80: return "A+" elif marks >= 70: return "A" elif marks >= 60: return "A-" elif marks >= 50: return "B" elif marks >= 40: return "C" else: return "F" marks = float(input("Marks: ")) grade = get_grade(marks) print("Grade:", grade)

try: a = float(input("প্রথম সংখ্যা: ")) b = float(input("দ্বিতীয় সংখ্যা: ")) print("যোগ:", a + b) print("বিয়োগ:", a - b) print("গুণ:", a * b) if b == 0: print("ভাগ করা সম্ভব নয়।") else: print("ভাগ:", a / b) except ValueError: print("দয়া করে valid number লিখুন।")

১৩. Class দিয়ে Bank Account

OOP শেখার জন্য বাস্তব জীবনের একটি সহজ example হলো bank account। Account-এর balance এবং deposit/withdraw method class-এর ভিতরে রাখা যায়।

class BankAccount: def __init__(self, owner, balance=0): self.owner = owner self.balance = balance def deposit(self, amount): self.balance = self.balance + amount print("Deposit successful.") def withdraw(self, amount): if amount <= self.balance: self.balance = self.balance - amount print("Withdrawal successful.") else: print("Insufficient balance.") def show_balance(self): print("Owner:", self.owner) print("Balance:", self.balance) account = BankAccount("Jakir", 5000) account.show_balance() account.deposit(2000) account.withdraw(1500) account.show_balance()
এখানে data এবং তার উপর কাজ করার method একই class-এর মধ্যে রাখা হয়েছে। এটিই OOP-এর একটি গুরুত্বপূর্ণ ধারণা।

১৪. Inheritance দিয়ে Employee System

একটি general Person class থেকে Employee class তৈরি করে common code reuse করা যায়।

class Person: def __init__(self, name): self.name = name def introduce(self): print("My name is", self.name) class Employee(Person): def __init__(self, name, salary): super().__init__(name) self.salary = salary def show_salary(self): print("Salary:", self.salary) employee = Employee("Jakir", 30000) employee.introduce() employee.show_salary()
super() parent class-এর constructor বা method ব্যবহার করতে সাহায্য করে।

import random secret = random.randint(1, 10) print("আমি 1 থেকে 10-এর মধ্যে একটি সংখ্যা ভেবেছি।") while True: guess = int(input("আপনার অনুমান: ")) if guess == secret: print("অভিনন্দন! আপনি সঠিক অনুমান করেছেন।") break elif guess < secret: print("আরও বড় সংখ্যা চেষ্টা করুন।") else: print("আরও ছোট সংখ্যা চেষ্টা করুন।")

১৭. Student Result — পূর্ণ Project

এবার কয়েকটি concept একসঙ্গে ব্যবহার করে একটি ছোট result management program তৈরি করা যাক।

name = input("Student name: ") subjects = ["Bangla", "English", "Math", "ICT", "Science"] marks = [] for subject in subjects: mark = float(input(subject + " marks: ")) marks.append(mark) total = sum(marks) average = total / len(marks) if average >= 80: grade = "A+" elif average >= 70: grade = "A" elif average >= 60: grade = "A-" elif average >= 50: grade = "B" elif average >= 40: grade = "C" else: grade = "F" print("\n===== RESULT =====") print("Name:", name) print("Total:", total) print("Average:", round(average, 2)) print("Highest:", max(marks)) print("Lowest:", min(marks)) print("Grade:", grade)
